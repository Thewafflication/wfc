## module Flow
Option Explicit

Public Function Inner(ByVal n As Long) As Long
    On Error GoTo h
    Inner = 10 \ n
    Exit Function
h:
    Inner = -1
End Function

Public Function Outer(ByVal n As Long) As Long
    On Error GoTo h
    Outer = 10 \ n
    Exit Function
h:
    Outer = Err.Number
    Resume Next
End Function

Public Function Rethrow() As Long
    On Error GoTo h
    Err.Raise 1234, "src", "first"
    Exit Function
h:
    Err.Raise Err.Number + 1, Err.Source, Err.Description & "+"
End Function

Public Function WithCleanup(ByVal fail As Boolean) As String
    Dim l As Life
    Set l = New Life
    l.Name = "c"
    If fail Then Err.Raise 77
    WithCleanup = "ok"
End Function

Public Function Recurse(ByVal n As Long) As Long
    If n = 0 Then

        Recurse = 1 + Recurse(n - 1)

        TakeVar = "Object"
    ElseIf IsNull(v) Then

        TakeVar = TypeName(v)

        TakeVar = TypeName(v) & ":" & v
    End If

Public Function PA(ParamArray p() As Variant) As String
    Dim i As Long
    Dim s As String
    s = LBound(p) & ":" & UBound(p)
    For i = LBound(p) To UBound(p)
        If IsMissing(p(i)) Then
            s = s & ";missing"
        ElseIf IsArray(p(i)) Then
            s = s & ";" & TypeName(p(i))
        ElseIf IsObject(p(i)) Then
            s = s & ";Object"
        ElseIf IsNull(p(i)) Then
            s = s & ";Null"
        Else
            s = s & ";" & TypeName(p(i)) & "=" & p(i)
        End If
    Next i
    PA = s
End Function
